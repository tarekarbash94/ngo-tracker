NGO & Donor Tracker · Setup
Connect your Google Sheet
Follow these three steps once, then your whole team can share the same live tool. Setup takes about 10 minutes.
1
Create the Google Sheet
Go to sheets.google.com → create a new spreadsheet → name it NGO Tracker.
Then create two sheets (tabs at the bottom): name them exactly NGOs and ActivityLog.

In the NGOs sheet, add these headers in row 1:
ID | Name | Type | Country | Focus | Capacity | Contact | Email | Website | Source | Priority | Notes | SubmittedBy | SubmittedDate | Status | Managers
In the ActivityLog sheet, add these headers in row 1:
ID | NgoID | Type | Description | LoggedBy | Date
2
Deploy a Google Apps Script as a Web App
In your Google Sheet, go to Extensions → Apps Script. Delete any existing code and paste this entire script, then click Deploy → New deployment → Web app. Set "Who has access" to Anyone and copy the Web App URL.
const SS = SpreadsheetApp.getActiveSpreadsheet(); function doGet(e) { const action = e.parameter.action; if (action === 'getNGOs') return res(getNGOs()); if (action === 'getLogs') return res(getLogs(e.parameter.ngoId)); return res({error:'unknown action'}); } function doPost(e) { const d = JSON.parse(e.postData.contents); if (d.action === 'addNGO') return res(addNGO(d)); if (d.action === 'updateStatus') return res(updateStatus(d)); if (d.action === 'addLog') return res(addLog(d)); return res({error:'unknown'}); } function res(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function getNGOs() { const sh = SS.getSheetByName('NGOs'); const rows = sh.getDataRange().getValues(); const headers = rows[0]; return rows.slice(1).map(r => { const o = {}; headers.forEach((h,i) => o[h] = r[i]); return o; }); } function getLogs(ngoId) { const sh = SS.getSheetByName('ActivityLog'); const rows = sh.getDataRange().getValues(); const headers = rows[0]; return rows.slice(1) .filter(r => String(r[1]) === String(ngoId)) .map(r => { const o = {}; headers.forEach((h,i) => o[h] = r[i]); return o; }); } function addNGO(d) { const sh = SS.getSheetByName('NGOs'); const id = Date.now(); sh.appendRow([id, d.Name, d.Type, d.Country, d.Focus, d.Capacity, d.Contact, d.Email, d.Website, d.Source, d.Priority, d.Notes, d.SubmittedBy, new Date().toLocaleDateString('en-GB'), 'pending', d.Managers||'']); return {success:true, id}; } function updateStatus(d) { const sh = SS.getSheetByName('NGOs'); const rows = sh.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(d.id)) { sh.getRange(i+1, 15).setValue(d.status); return {success:true}; } } return {error:'not found'}; } function addLog(d) { const sh = SS.getSheetByName('ActivityLog'); const id = Date.now(); sh.appendRow([id, d.ngoId, d.type, d.description, d.loggedBy, new Date().toLocaleDateString('en-GB')]); return {success:true, id}; }
⚠ After deploying, authorise the script when Google asks. Copy the Web App URL — it looks like: https://script.google.com/macros/s/XXXX.../exec
